import GridMount from "./GridMount";
import Grid, { PivotDataViewModel } from "grid/dist/renderer";

# Getting Started

The grid library renders high-performance pivot tables in the browser.
Below is a live demo with default sample data — four quarters across two regions and channels.

<GridMount setup={(el) => {
  const grid = new Grid({}, el);
  grid.data = new PivotDataViewModel(
    [[1200, 800, 1400, 950], [3500, 2100, 4200, 3100], [7200, 5300, 6800, 4900], [920, 680, 1050, 870]],
    [["Q1", "Q2", "Q3", "Q4"]],
    [["North", "North", "South", "South"], ["Online", "Retail", "Online", "Retail"]],
  );
  grid.draw();
}} />

## Custom Facets

You can pass multi-level column and row facets to control the pivot axes.

<GridMount height="250px" setup={(el) => {
  const grid = new Grid({}, el);
  grid.data = new PivotDataViewModel(
    [[5200, 3800, 6100, 4400], [4800, 3200, 5500, 4100], [6700, 5100, 7200, 5900], [3900, 2700, 4300, 3500]],
    [["2023", "2023", "2024", "2024"], ["H1", "H2", "H1", "H2"]],
    [["East", "East", "West", "West"], ["Wholesale", "Direct", "Wholesale", "Direct"]],
  );
  grid.draw();
}} />

## Providing Data

Pass a column-major `data` array (each inner array is one column of cell values):

<GridMount height="160px" setup={(el) => {
  const grid = new Grid({}, el);
  grid.data = new PivotDataViewModel(
    [[10, 20], [30, 40], [50, 60]],
    [["A", "B", "C"]],
    [["Row 1", "Row 2"]],
  );
  grid.draw();
}} />
